cmake_minimum_required(VERSION 4.4)
project(ftz_api_examples LANGUAGES CXX)
find_package(ftz CONFIG REQUIRED COMPONENTS ftz)
enable_testing()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
  set(API_PROFILES NEON CACHE STRING "Admitted example architectures")
else()
  set(API_PROFILES AVX2 CACHE STRING "Admitted example architectures")
endif()
foreach(name scalar environment)
  add_executable(api_${name} ${name}.cc)
  target_link_libraries(api_${name} PRIVATE ftz::ftz)
  set_target_properties(api_${name} PROPERTIES CXX_SCAN_FOR_MODULES ON)
  add_test(NAME api.${name} COMMAND api_${name})
endforeach()
foreach(profile IN LISTS API_PROFILES)
  string(TOLOWER "${profile}" name)
  if(profile STREQUAL "AVX512")
    set(bits 512)
  elseif(profile STREQUAL "NEON")
    set(bits 128)
  elseif(profile STREQUAL "AVX2")
    set(bits 256)
  else()
    message(FATAL_ERROR "Select AVX2, AVX512 or NEON")
  endif()
  add_executable(api_vectors_${name} vectors.cc)
  target_link_libraries(api_vectors_${name} PRIVATE ftz::ftz simd::common simd::${name})
  simd_target_profile(api_vectors_${name} "${profile}")
  target_compile_definitions(api_vectors_${name} PRIVATE API_PROFILE=${bits})
  set_target_properties(api_vectors_${name} PROPERTIES CXX_SCAN_FOR_MODULES ON)
  add_test(NAME api.vectors.${name} COMMAND api_vectors_${name})
endforeach()
