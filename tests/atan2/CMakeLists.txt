cmake_minimum_required(VERSION 4.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(ftz_native_atan2 LANGUAGES CXX)
  find_package(ftz CONFIG REQUIRED COMPONENTS ftz)
endif()
enable_testing()
if(DEFINED FTZ_TEST_PROFILES)
  set(ATAN2_PROFILES "${FTZ_TEST_PROFILES}")
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
  set(ATAN2_PROFILES NEON CACHE STRING "Admitted test architectures")
else()
  set(ATAN2_PROFILES AVX2 CACHE STRING "Admitted test architectures")
endif()
foreach(profile IN LISTS ATAN2_PROFILES)
 string(TOLOWER "${profile}" name)
 if(profile STREQUAL "AVX512")
  set(bits 512)
 elseif(profile STREQUAL "NEON")
  set(bits 128)
 else()
  set(bits 256)
 endif()
 add_executable(atan2_${name} atan2.cc)
 add_library(atan2_codegen_${name} OBJECT codegen.cc)
 foreach(target IN ITEMS atan2_${name} atan2_codegen_${name})
  target_link_libraries(${target} PRIVATE ftz::ftz simd::common simd::${name})
  simd_target_profile(${target} "${profile}")
  target_compile_definitions(${target} PRIVATE FTZ_TEST_PROFILE=${bits})
  if(MSVC)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
  endif()
  if(COMMAND ftz_link_clang_cl_asan)
    ftz_link_clang_cl_asan(${target})
  endif()
  set_target_properties(${target} PROPERTIES CXX_SCAN_FOR_MODULES ON INTERPROCEDURAL_OPTIMIZATION FALSE)
 endforeach()
 add_test(NAME atan2.${name} COMMAND atan2_${name})
endforeach()
