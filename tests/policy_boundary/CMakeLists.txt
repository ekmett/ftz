cmake_minimum_required(VERSION 4.4)
project(ftz_policy_boundary LANGUAGES CXX)
find_package(ftz CONFIG REQUIRED COMPONENTS ftz)
enable_testing()
option(REVIEW_BASELINE "Compile codegen against the older single-policy API" OFF)
set(REVIEW_PROFILES "AVX2;AVX512" CACHE STRING "Profiles available on this host")
foreach(profile IN LISTS REVIEW_PROFILES)
  if(NOT profile MATCHES "^(AVX2|AVX512)$")
    message(FATAL_ERROR "This bounded codegen fixture covers AVX2 and AVX512")
  endif()
  string(TOLOWER "${profile}" name)
  add_library(codegen_${name} OBJECT codegen.cc)
  add_library(codegen_array_${name} OBJECT codegen_array.cc)
  set(targets codegen_${name} codegen_array_${name})
  if(NOT REVIEW_BASELINE)
    add_executable(policy_${name} policy.cc)
    add_executable(surface_${name} surface.cc)
    list(APPEND targets policy_${name} surface_${name})
    add_test(NAME policy.${name} COMMAND policy_${name})
    add_test(NAME surface.${name} COMMAND surface_${name})
  endif()
  foreach(target IN LISTS targets)
    target_link_libraries(${target} PRIVATE ftz::ftz native::common native::${name})
    native_target_profile(${target} "${profile}")
    set_target_properties(${target} PROPERTIES CXX_SCAN_FOR_MODULES ON INTERPROCEDURAL_OPTIMIZATION FALSE)
    target_compile_definitions(${target} PRIVATE REVIEW_AVX512=$<STREQUAL:${profile},AVX512> REVIEW_BASELINE=$<BOOL:${REVIEW_BASELINE}>)
  endforeach()
endforeach()
