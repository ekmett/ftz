set(FTZ_RAW_SCALEB_FTZ_PROFILE "before-rounding" CACHE STRING
  "Captured raw scaling tininess: before-rounding or after-rounding")
set_property(CACHE FTZ_RAW_SCALEB_FTZ_PROFILE PROPERTY STRINGS before-rounding after-rounding)
if(NOT FTZ_RAW_SCALEB_FTZ_PROFILE MATCHES "^(before-rounding|after-rounding)$")
  message(FATAL_ERROR "Select a captured raw scaling FTZ profile")
endif()
string(TOLOWER "${FTZ_TEST_ISA}" profile)
if(FTZ_TEST_ISA STREQUAL "AVX512")
  set(profile_bits 512)
elseif(FTZ_TEST_ISA STREQUAL "AVX2")
  set(profile_bits 256)
elseif(FTZ_TEST_ISA STREQUAL "NEON")
  set(profile_bits 128)
else()
  message(FATAL_ERROR "FTZ regression fixtures require AVX2, AVX512 or NEON")
endif()
foreach(suite IN ITEMS ftz32_simd scaleb noexcept_ftz module_contract)
  set(target ftz_test_${suite})
  add_executable(${target} ${suite}.cc)
  target_link_libraries(${target} PRIVATE ftz::ftz simd::${profile} simd::common)
  simd_target_profile(${target} "${FTZ_TEST_ISA}")
  target_compile_definitions(${target} PRIVATE FTZ_TEST_PROFILE=${profile_bits})
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  set_target_properties(${target} PROPERTIES CXX_SCAN_FOR_MODULES ON)
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(${target} PRIVATE /W4 /WX /wd4127 /wd4702)
    # Object-format flags do not change imported module interfaces.
    set_property(TARGET ${target} APPEND_STRING PROPERTY COMPILE_FLAGS " /bigobj")
    target_link_options(${target} PRIVATE /STACK:16777216)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    if(WIN32)
      target_link_options(${target} PRIVATE -Xlinker /STACK:16777216)
    endif()
  endif()
  if(COMMAND ftz_link_clang_cl_asan)
    ftz_link_clang_cl_asan(${target})
  endif()
endforeach()
foreach(mode IN ITEMS gradual flush)
  add_test(NAME ftz.simd.${mode} COMMAND ftz_test_ftz32_simd ${mode})
  add_test(NAME ftz.scaleb.${mode}
    COMMAND ftz_test_scaleb ${mode} ${FTZ_RAW_SCALEB_FTZ_PROFILE})
endforeach()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
  foreach(mode IN ITEMS daz ftz)
    add_test(NAME ftz.scaleb.${mode}
      COMMAND ftz_test_scaleb ${mode} ${FTZ_RAW_SCALEB_FTZ_PROFILE})
  endforeach()
endif()
add_test(NAME ftz.noexcept COMMAND ftz_test_noexcept_ftz)
add_test(NAME ftz.consumer.capture
  COMMAND ftz_test_module_contract "${CMAKE_CURRENT_BINARY_DIR}/consumer.bin")
get_property(regression_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${regression_tests} PROPERTIES TIMEOUT 180 LABELS "host;math;exact")

# New shapes are tested independently for every configured ISA provider.
foreach(isa IN LISTS FTZ_TEST_PROFILES)
  string(TOLOWER "${isa}" profile)
  if(isa STREQUAL "AVX512")
    set(profile_bits 512)
  elseif(isa STREQUAL "AVX2")
    set(profile_bits 256)
  elseif(isa STREQUAL "NEON")
    set(profile_bits 128)
  else()
    continue()
  endif()
  set(target ftz_test_swizzle_${profile})
  add_executable(${target} swizzle.cc)
  target_link_libraries(${target} PRIVATE ftz::ftz simd::${profile} simd::common)
  simd_target_profile(${target} "${isa}")
  target_compile_definitions(${target} PRIVATE FTZ_TEST_PROFILE=${profile_bits})
  set_target_properties(${target} PROPERTIES CXX_SCAN_FOR_MODULES ON)
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(${target} PRIVATE /W4 /WX)
    # Object-format flags do not change imported module interfaces.
    set_property(TARGET ${target} APPEND_STRING PROPERTY COMPILE_FLAGS " /bigobj")
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
  if(COMMAND ftz_link_clang_cl_asan)
    ftz_link_clang_cl_asan(${target})
  endif()
  set(dual ftz_test_dual_${profile})
  add_executable(${dual} dual_policy.cc)
  target_link_libraries(${dual} PRIVATE ftz::ftz simd::${profile} simd::common)
  simd_target_profile(${dual} "${isa}")
  target_compile_definitions(${dual} PRIVATE FTZ_TEST_PROFILE=${profile_bits})
  set_target_properties(${dual} PROPERTIES CXX_SCAN_FOR_MODULES ON INTERPROCEDURAL_OPTIMIZATION FALSE)
  if(COMMAND ftz_link_clang_cl_asan)
    ftz_link_clang_cl_asan(${dual})
  endif()
  add_test(NAME ftz.dual.${profile} COMMAND ${dual})
  set_tests_properties(ftz.dual.${profile} PROPERTIES TIMEOUT 180 LABELS "host;math;exact;policies")
  set(wide_exp ftz_test_wide_exp_${profile})
  add_executable(${wide_exp} wide_exp.cc)
  target_link_libraries(${wide_exp} PRIVATE ftz::ftz simd::${profile} simd::common)
  simd_target_profile(${wide_exp} "${isa}")
  target_compile_definitions(${wide_exp} PRIVATE FTZ_TEST_PROFILE=${profile_bits})
  set_target_properties(${wide_exp} PROPERTIES CXX_SCAN_FOR_MODULES ON INTERPROCEDURAL_OPTIMIZATION FALSE)
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_options(${wide_exp} PRIVATE /W4 /WX)
    set_property(TARGET ${wide_exp} APPEND_STRING PROPERTY COMPILE_FLAGS " /bigobj")
  else()
    target_compile_options(${wide_exp} PRIVATE -Wall -Wextra -Werror)
  endif()
  if(COMMAND ftz_link_clang_cl_asan)
    ftz_link_clang_cl_asan(${wide_exp})
  endif()
  add_test(NAME ftz.wide_exp.${profile} COMMAND ${wide_exp})
  set_tests_properties(ftz.wide_exp.${profile} PROPERTIES TIMEOUT 180 LABELS "host;math;exact;policies;wide")
  foreach(mode IN ITEMS gradual flush)
    add_test(NAME ftz.swizzle.${profile}.${mode} COMMAND ${target} ${mode})
    set_tests_properties(ftz.swizzle.${profile}.${mode} PROPERTIES TIMEOUT 180 LABELS "host;math;exact;swizzle")
  endforeach()
endforeach()
