# Controls and admission use the configured SIMD minimum, without an extra ISA provider.
foreach(suite IN ITEMS controls admission)
  set(target ftz_test_${suite})
  add_executable(${target} ${suite}.cc)
  target_link_libraries(${target} PRIVATE ftz::ftz)
  set_target_properties(${target} PROPERTIES CXX_SCAN_FOR_MODULES ON
    INTERPROCEDURAL_OPTIMIZATION FALSE)
  if(COMMAND ftz_link_clang_cl_asan)
    ftz_link_clang_cl_asan(${target})
  endif()
  add_test(NAME ftz.${suite} COMMAND ${target})
  set_tests_properties(ftz.${suite} PROPERTIES LABELS "host;modules;ftz;boundary" TIMEOUT 180)
endforeach()

# Ordinary caller and IPO caller remain separate; neither runs admission.
set(entry_modes ordinary)
if(FTZ_ENABLE_IPO)
  list(APPEND entry_modes ipo)
endif()
foreach(mode IN LISTS entry_modes)
  set(target ftz_test_entry_${mode})
  add_executable(${target} thread_entry.cc)
  target_link_libraries(${target} PRIVATE ftz::ftz)
  if(mode STREQUAL "ipo")
    set(ipo TRUE)
  else()
    set(ipo FALSE)
  endif()
  set_target_properties(${target} PROPERTIES CXX_SCAN_FOR_MODULES ON
    INTERPROCEDURAL_OPTIMIZATION ${ipo})
  if(COMMAND ftz_link_clang_cl_asan)
    ftz_link_clang_cl_asan(${target})
  endif()
  add_test(NAME ftz.entry.${mode} COMMAND ${target})
  set_tests_properties(ftz.entry.${mode} PROPERTIES LABELS "host;modules;ftz;boundary" TIMEOUT 180)
endforeach()

string(TOLOWER "${FTZ_TEST_ISA}" profile)
if(FTZ_TEST_ISA STREQUAL "AVX512")
  set(profile_bits 512)
elseif(FTZ_TEST_ISA STREQUAL "AVX2")
  set(profile_bits 256)
else()
  set(profile_bits 128)
endif()
add_executable(ftz_test_consumer consumer.cc)
target_link_libraries(ftz_test_consumer PRIVATE ftz::ftz native::common native::${profile})
native_target_profile(ftz_test_consumer "${FTZ_TEST_ISA}")
target_compile_definitions(ftz_test_consumer PRIVATE FTZ_TEST_PROFILE=${profile_bits})
set_target_properties(ftz_test_consumer PROPERTIES CXX_SCAN_FOR_MODULES ON)
if(COMMAND ftz_link_clang_cl_asan)
  ftz_link_clang_cl_asan(ftz_test_consumer)
endif()
add_test(NAME ftz.consumer COMMAND ftz_test_consumer)
set_tests_properties(ftz.consumer PROPERTIES LABELS "host;modules;ftz" TIMEOUT 180)

# Compile identity and independent custom elements through both ISA tags. No runtime
# per-profile admission or ISA-sensitive entry point is added by this fixture.
if("AVX2" IN_LIST FTZ_TEST_PROFILES AND "AVX512" IN_LIST FTZ_TEST_PROFILES)
  add_library(ftz_test_identity OBJECT identity.cc custom_elements.cc)
  target_link_libraries(ftz_test_identity PRIVATE ftz::ftz native::common native::avx2 native::avx512)
  native_target_profile(ftz_test_identity AVX512)
  set_target_properties(ftz_test_identity PROPERTIES CXX_SCAN_FOR_MODULES ON)
endif()
