if("NEON" IN_LIST FTZ_TEST_PROFILES)
  set(profiles NEON)
elseif("AVX2" IN_LIST FTZ_TEST_PROFILES AND "AVX512" IN_LIST FTZ_TEST_PROFILES)
  set(profiles AVX2 AVX512)
else()
  return()
endif()
foreach(profile IN LISTS profiles)
  string(TOLOWER "${profile}" lower)
  if(profile STREQUAL "AVX2")
    set(bits 256)
  elseif(profile STREQUAL "AVX512")
    set(bits 512)
  else()
    set(bits 128)
  endif()
  set(target ftz_profile_${lower})
  add_library(${target} OBJECT bridge.cc)
  target_link_libraries(${target} PRIVATE ftz::ftz native::common native::${lower})
  native_target_profile(${target} ${profile})
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../core_regression")
  target_compile_definitions(${target} PRIVATE FTZ_TEST_PROFILE=${bits} FTZ_TEST_ENTRY=profile_${lower})
  set_target_properties(${target} PROPERTIES CXX_SCAN_FOR_MODULES ON INTERPROCEDURAL_OPTIMIZATION FALSE)
  list(APPEND profile_objects $<TARGET_OBJECTS:${target}>)
  list(APPEND profile_libraries native::${lower})
endforeach()
if("NEON" IN_LIST profiles)
  set(main neon.cc)
else()
  set(main main.cc)
endif()
add_executable(ftz_test_profiles ${main} ${profile_objects})
# TARGET_OBJECTS does not carry the profile objects' link dependencies.
target_link_libraries(ftz_test_profiles PRIVATE ftz::ftz ${profile_libraries})
set_target_properties(ftz_test_profiles PROPERTIES CXX_SCAN_FOR_MODULES ON INTERPROCEDURAL_OPTIMIZATION FALSE)
if(COMMAND ftz_link_clang_cl_asan)
  ftz_link_clang_cl_asan(ftz_test_profiles)
endif()
if("NEON" IN_LIST profiles)
  add_test(NAME ftz.profiles.baseline COMMAND ftz_test_profiles
    "${CMAKE_CURRENT_BINARY_DIR}/profiles.bin" "${CMAKE_CURRENT_SOURCE_DIR}/baseline.bin")
else()
  foreach(mode IN ITEMS all avx2 avx512 none)
    add_test(NAME ftz.profiles.${mode} COMMAND ftz_test_profiles ${mode})
    set_tests_properties(ftz.profiles.${mode} PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 180 LABELS "host;modules;profiles;exact")
  endforeach()
  add_test(NAME ftz.profiles.baseline COMMAND ftz_test_profiles all
    "${CMAKE_CURRENT_BINARY_DIR}/profiles.bin" "${CMAKE_CURRENT_SOURCE_DIR}/baseline.bin")
endif()
set_tests_properties(ftz.profiles.baseline PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 180 LABELS "host;modules;profiles;exact")
