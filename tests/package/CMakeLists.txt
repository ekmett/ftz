cmake_minimum_required(VERSION 4.4)
project(ftz_package_consumer LANGUAGES CXX)
find_package(ftz CONFIG REQUIRED COMPONENTS ftz)
set(TEST_ISA AVX2 CACHE STRING "Native profile used by this independent consumer")
option(TEST_IPO "Use IPO across all three static libraries" ON)
option(TEST_PCH "Compile the consumer module with its own PCH" ON)
string(TOLOWER "${TEST_ISA}" profile)
if(NOT TARGET native::${profile})
  message(FATAL_ERROR "The installed simd package lacks ${TEST_ISA}.")
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
configure_file(package.ccm.in "${CMAKE_CURRENT_BINARY_DIR}/package.ccm" @ONLY)
add_library(package STATIC implementation.cc)
target_sources(package PUBLIC FILE_SET cxx_modules TYPE CXX_MODULES
  BASE_DIRS "${CMAKE_CURRENT_BINARY_DIR}" FILES "${CMAKE_CURRENT_BINARY_DIR}/package.ccm")
target_link_libraries(package PUBLIC ftz::ftz native::${profile})
native_target_profile(package "${TEST_ISA}")
set_target_properties(package PROPERTIES CXX_SCAN_FOR_MODULES ON INTERPROCEDURAL_OPTIMIZATION ${TEST_IPO})
if(TEST_PCH)
  target_precompile_headers(package PRIVATE <array> <bit> <cstdint> <native/attributes.h>)
endif()
add_executable(consumer main.cc)
target_link_libraries(consumer PRIVATE package)
native_target_profile(consumer "${TEST_ISA}")
set_target_properties(consumer PROPERTIES CXX_SCAN_FOR_MODULES ON INTERPROCEDURAL_OPTIMIZATION ${TEST_IPO})
enable_testing()
add_test(NAME package.transitive COMMAND consumer)
