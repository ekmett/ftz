foreach(isa IN LISTS FTZ_TEST_PROFILES)
  string(TOLOWER "${isa}" profile)
  if(isa STREQUAL "AVX2")
    set(bits 256)
  elseif(isa STREQUAL "AVX512")
    set(bits 512)
  elseif(isa STREQUAL "NEON")
    set(bits 128)
  else()
    continue()
  endif()
  set(target ftz_test_tanh_${profile})
  add_executable(${target} tanh.cc)
  target_link_libraries(${target} PRIVATE ftz::ftz simd::common simd::${profile})
  simd_target_profile(${target} ${isa})
  target_compile_definitions(${target} PRIVATE FTZ_TEST_PROFILE=${bits})
  set_target_properties(${target} PROPERTIES CXX_SCAN_FOR_MODULES ON INTERPROCEDURAL_OPTIMIZATION FALSE)
  if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(${target} PRIVATE /W4 /WX /bigobj)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
  if(COMMAND ftz_link_clang_cl_asan)
    ftz_link_clang_cl_asan(${target})
  endif()
  add_test(NAME ftz.tanh.${profile} COMMAND ${target})
  set_tests_properties(ftz.tanh.${profile} PROPERTIES TIMEOUT 180 LABELS "host;math;exact;tanh")
endforeach()
