cmake_minimum_required(VERSION 4.4)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(ftz_native_tests LANGUAGES CXX)
  find_package(ftz CONFIG REQUIRED COMPONENTS ftz)
endif()
enable_testing()
if(DEFINED FTZ_TEST_PROFILES)
  set(LOG_PROFILES "${FTZ_TEST_PROFILES}")
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
  set(LOG_PROFILES NEON CACHE STRING "Admitted test architectures")
else()
  set(LOG_PROFILES AVX2 CACHE STRING "Admitted test architectures")
endif()
option(FTZ_BUILD_MPFR_TESTS "Build optional independent MPFR log/log1p accuracy tests" OFF)
if(FTZ_BUILD_MPFR_TESTS)
  find_path(FTZ_MPFR_INCLUDE_DIR mpfr.h)
  find_path(FTZ_GMP_INCLUDE_DIR gmp.h)
  find_library(FTZ_MPFR_LIBRARY NAMES mpfr)
  find_library(FTZ_GMP_LIBRARY NAMES gmp)
  if(NOT FTZ_MPFR_INCLUDE_DIR OR NOT FTZ_GMP_INCLUDE_DIR OR NOT FTZ_MPFR_LIBRARY OR NOT FTZ_GMP_LIBRARY)
    message(FATAL_ERROR "FTZ_BUILD_MPFR_TESTS=ON requires installed MPFR and GMP headers/libraries. Set CMAKE_PREFIX_PATH or FTZ_MPFR_INCLUDE_DIR, FTZ_GMP_INCLUDE_DIR, FTZ_MPFR_LIBRARY and FTZ_GMP_LIBRARY; nothing is downloaded automatically.")
  endif()
endif()
foreach(profile IN LISTS LOG_PROFILES)
 string(TOLOWER "${profile}" name)
 if(profile STREQUAL "AVX512")
  set(bits 512)
 elseif(profile STREQUAL "NEON")
  set(bits 128)
 else()
  set(bits 256)
 endif()
 add_executable(log_${name} log.cc)
 add_library(log_codegen_${name} OBJECT codegen.cc)
 foreach(target IN ITEMS log_${name} log_codegen_${name})
  target_link_libraries(${target} PRIVATE ftz::ftz simd::common simd::${name})
  if(MSVC)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
  endif()
  simd_target_profile(${target} "${profile}")
  target_compile_definitions(${target} PRIVATE FTZ_TEST_PROFILE=${bits})
  if(COMMAND ftz_link_clang_cl_asan)
    ftz_link_clang_cl_asan(${target})
  endif()
  set_target_properties(${target} PROPERTIES CXX_SCAN_FOR_MODULES ON INTERPROCEDURAL_OPTIMIZATION FALSE)
 endforeach()
 add_test(NAME log.${name} COMMAND log_${name})
endforeach()

if(FTZ_BUILD_MPFR_TESTS)
  foreach(profile IN LISTS LOG_PROFILES)
    string(TOLOWER "${profile}" name)
    if(profile STREQUAL "AVX512")
      set(bits 512)
    elseif(profile STREQUAL "NEON")
      set(bits 128)
    else()
      set(bits 256)
    endif()
    add_executable(log_mpfr_${name} mpfr.cc)
    target_link_libraries(log_mpfr_${name} PRIVATE ftz::ftz simd::common simd::${name}
      "${FTZ_MPFR_LIBRARY}" "${FTZ_GMP_LIBRARY}")
    target_include_directories(log_mpfr_${name} PRIVATE "${FTZ_MPFR_INCLUDE_DIR}" "${FTZ_GMP_INCLUDE_DIR}")
    target_compile_definitions(log_mpfr_${name} PRIVATE FTZ_TEST_PROFILE=${bits})
    simd_target_profile(log_mpfr_${name} "${profile}")
    set_target_properties(log_mpfr_${name} PROPERTIES CXX_SCAN_FOR_MODULES ON INTERPROCEDURAL_OPTIMIZATION FALSE)
    if(COMMAND ftz_link_clang_cl_asan)
      ftz_link_clang_cl_asan(log_mpfr_${name})
    endif()
    add_test(NAME log.mpfr.${name} COMMAND log_mpfr_${name} "${CMAKE_CURRENT_BINARY_DIR}/mpfr-${name}.txt")
    set_tests_properties(log.mpfr.${name} PROPERTIES LABELS "mpfr;accuracy")
  endforeach()
endif()
